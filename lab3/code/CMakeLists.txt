cmake_minimum_required(VERSION 3.24)
project(CampusMarket VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "CampusMarket uses native Windows GUI, CNG and DPAPI.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(SQLITE_SOURCE_DIR "" CACHE PATH "Offline directory containing sqlite3.c and sqlite3.h")
if(NOT SQLITE_SOURCE_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sqlite/sqlite3.c")
  set(SQLITE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sqlite")
endif()
if(NOT SQLITE_SOURCE_DIR)
  FetchContent_Declare(sqlite
    URL https://www.sqlite.org/2026/sqlite-amalgamation-3540100.zip
    URL_HASH SHA3_256=92281c86e2957f9b9305890f9ac04204caa7b677cc868dc326d942689df28e30
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(sqlite)
  set(SQLITE_SOURCE_DIR ${sqlite_SOURCE_DIR})
endif()
add_library(sqlite STATIC ${SQLITE_SOURCE_DIR}/sqlite3.c)
target_include_directories(sqlite PUBLIC ${SQLITE_SOURCE_DIR})
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0)
add_library(market_core src/database.cpp src/security.cpp src/service.cpp)
target_include_directories(market_core PUBLIC include)
target_link_libraries(market_core PUBLIC sqlite bcrypt crypt32)
target_compile_definitions(market_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_executable(CampusMarket WIN32 src/main.cpp src/ui.cpp app.rc)
target_link_libraries(CampusMarket PRIVATE market_core comctl32 shell32 user32 gdi32 ole32)
if(MINGW)
  target_link_options(CampusMarket PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
foreach(target market_core CampusMarket)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(market_tests tests/service_test.cpp)
  target_link_libraries(market_tests PRIVATE market_core)
  target_compile_options(market_tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  if(MINGW)
    target_link_options(market_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME business_rules COMMAND market_tests)
  set_tests_properties(business_rules PROPERTIES TIMEOUT 180)
endif()
option(MARKET_UI_SMOKE "Build isolated demo UI harness; never use real data" OFF)
if(MARKET_UI_SMOKE)
  add_executable(market_ui_smoke WIN32 src/main.cpp src/ui.cpp app.rc)
  target_compile_definitions(market_ui_smoke PRIVATE MARKET_UI_SMOKE)
  target_compile_options(market_ui_smoke PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  target_link_libraries(market_ui_smoke PRIVATE market_core comctl32 shell32 user32 gdi32 ole32)
  if(MINGW)
    target_link_options(market_ui_smoke PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()
install(TARGETS CampusMarket RUNTIME DESTINATION .)
